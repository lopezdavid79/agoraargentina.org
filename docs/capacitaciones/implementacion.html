1. parseGrabaciones — parser del hidden-JSON

Función pura que convierte el contenido del input oculto grabaciones_json en un arreglo normalizado de { url, label }. Comportamiento:

  • Parsea JSON.parse(body.grabaciones_json || '[]').
  • Elementos string → { url: s.trim(), label: '' } (legacy).
  • Elementos objeto → url y label recortados vía String(...).trim().
  • Descarta elementos sin url tras recortar y limita a 10 con .slice(0, 10).
  • JSON inválido → devuelve [] (catch).
function parseGrabaciones(body) {
    try {
        const raw = JSON.parse(body.grabaciones_json || '[]');
        return raw.map(r => {
            if (typeof r === 'string') {
                return { url: r.trim(), label: '' };
            }
            return {
                url: String((r && r.url) || '').trim(),
                label: String((r && r.label) || '').trim()
            };
        }).filter(g => g.url.length > 0).slice(0, 10);
    } catch { return []; }
}
